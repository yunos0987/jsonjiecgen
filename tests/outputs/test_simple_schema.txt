TEST_S(test_SimpleSchema_jsontext_to_iecdata)
	var
		txt: string[JSON_STRING_SIZE];
		data: SimpleSchema_t;
		txt1: string[JSON_STRING_SIZE];
		data1: SimpleSchema_t;
		txt2: string[JSON_STRING_SIZE];
		data2: SimpleSchema_t;
		txt3: string[JSON_STRING_SIZE];
		data3: SimpleSchema_t;
		txt4: string[JSON_STRING_SIZE];
		data4: SimpleSchema_t;
		txt5: string[JSON_STRING_SIZE];
		data5: SimpleSchema_t;
		txt6: string[JSON_STRING_SIZE];
		data6: SimpleSchema_t;
	end
	{st}
	txt := '{"name":"","value":0.0,"count":0,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt, data);
	EXPECT_EQ_STRING('', data.name);
	EXPECT_NEAR(0.0, data.value, 0.001);
	EXPECT_EQ_DINT(0, data.count);
	EXPECT_FALSE(data.active);
	txt1 := '{"name":"hello\"world","value":0.0,"count":0,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt1, data1);
	EXPECT_EQ_STRING('hello"world', data1.name);
	txt2 := '{"name":"","value":0.0,"count":0,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt2, data2);
	EXPECT_EQ_STRING('', data2.name);
	txt3 := '{"name":"","value":42,"count":0,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt3, data3);
	EXPECT_NEAR(42.0, data3.value, 0.001);
	txt4 := '{"name":"","value":1e3,"count":0,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt4, data4);
	EXPECT_NEAR(1000.0, data4.value, 0.001);
	txt5 := '{"name":"","value":-1.5,"count":0,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt5, data5);
	EXPECT_NEAR(-1.5, data5.value, 0.001);
	txt6 := '{"name":"","value":0.0,"count":-42,"active":false}';
	SimpleSchema_jsontext_to_iecdata(txt6, data6);
	EXPECT_EQ_DINT(-42, data6.count);
	{end}
END_TEST_S

		data5_rt: SimpleSchema_t;
		data6: SimpleSchema_t;

		data6_rt: SimpleSchema_t;
	end
	{st}
	data.name := '';
	data.value := 0.0;
	data.count := 0;
	data.active := false;
	SimpleSchema_iecdata_to_jsontext(data, txt);

	EXPECT_NEAR(0.0, data_rt.value, 0.001);
	EXPECT_EQ_DINT(0, data_rt.count);
	EXPECT_FALSE(data_rt.active);
	data1.name := 'hello"world';

	SimpleSchema_jsontext_to_iecdata(txt1, data1_rt);
	EXPECT_EQ_STRING('hello"world', data1_rt.name);
	data2.name := '';
	SimpleSchema_iecdata_to_jsontext(data2, txt2);
	SimpleSchema_jsontext_to_iecdata(txt2, data2_rt);
	EXPECT_EQ_STRING('', data2_rt.name);
	data3.value := 42.0;
	SimpleSchema_iecdata_to_jsontext(data3, txt3);
	SimpleSchema_jsontext_to_iecdata(txt3, data3_rt);
	EXPECT_NEAR(42.0, data3_rt.value, 0.001);
	data4.value := 1000.0;
	SimpleSchema_iecdata_to_jsontext(data4, txt4);
	SimpleSchema_jsontext_to_iecdata(txt4, data4_rt);
	EXPECT_NEAR(1000.0, data4_rt.value, 0.001);
	data5.value := -1.5;
	SimpleSchema_iecdata_to_jsontext(data5, txt5);
	SimpleSchema_jsontext_to_iecdata(txt5, data5_rt);
	EXPECT_NEAR(-1.5, data5_rt.value, 0.001);
	data6.count := -42;
	SimpleSchema_iecdata_to_jsontext(data6, txt6);
	SimpleSchema_jsontext_to_iecdata(txt6, data6_rt);
	EXPECT_EQ_DINT(-42, data6_rt.count);
	{end}

TEST_S(test_json_bytes_find_key)
	var
		raw: array[0..127] of byte;
		key_x: array[0..0] of byte := [byte#120];
		key_b: array[0..0] of byte := [byte#98];
		key_key: array[0..2] of byte := [byte#107, byte#101, byte#121];
		wp: dint;
		idx: dint;
		ok: bool;
	end
	{st}
	// basic: single integer field
	// {"x":42} - key 'x' value starts at byte 5 ('4' = 52)
	wp := json_bytes_insert_text(dat:=raw, index:=0, str:='{"x":42}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(dat:=raw, key:=key_x, start_index:=0, end_index:=-1, eno=>ok);
	EXPECT_EQ_DINT(5, idx);
	EXPECT_EQ_DINT(52, byte_to_dint(raw[idx]));
	// second field: {"a":1,"b":2} - key 'b' value starts at byte 11 ('2' = 50)
	wp := json_bytes_insert_text(dat:=raw, index:=0, str:='{"a":1,"b":2}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(dat:=raw, key:=key_b, start_index:=0, end_index:=-1, eno=>ok);
	EXPECT_EQ_DINT(11, idx);
	EXPECT_EQ_DINT(50, byte_to_dint(raw[idx]));
	// nested: key inside nested object is not matched; top-level key is
	// {"a":{"key":1},"key":2} - top-level 'key' value at byte 21 ('2' = 50)
	wp := json_bytes_insert_text(dat:=raw, index:=0, str:='{"a":{"key":1},"key":2}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(dat:=raw, key:=key_key, start_index:=0, end_index:=-1, eno=>ok);
	EXPECT_EQ_DINT(21, idx);
	EXPECT_EQ_DINT(50, byte_to_dint(raw[idx]));
	// not found: returns -1
	wp := json_bytes_insert_text(dat:=raw, index:=0, str:='{"other":1}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(dat:=raw, key:=key_x, start_index:=0, end_index:=-1, eno=>ok);
	EXPECT_EQ_DINT(-1, idx);
	{end}
END_TEST_S

TEST_S(test_json_find_block)
	var
		src: string[JSON_STRING_SIZE];
		blk: string[JSON_STRING_SIZE];
		ok: bool;
	end
	{st}
	// basic: extract object block
	src := '{"key":{"a":1},"other":2}';
	json_find_block(dat:=src, key:='key', open_ch:='{', close_ch:='}', block:=blk, eno=>ok);
	EXPECT_TRUE(ok);
	EXPECT_EQ_STRING('{"a":1}', blk);
	// extract array block
	src := '{"arr":[1,2,3],"x":0}';
	json_find_block(dat:=src, key:='arr', open_ch:='[', close_ch:=']', block:=blk, eno=>ok);
	EXPECT_TRUE(ok);
	EXPECT_EQ_STRING('[1,2,3]', blk);
	// nested: outer block returned (ignores inner brackets of same type)
	src := '{"a":{"b":{"c":1}}}';
	json_find_block(dat:=src, key:='a', open_ch:='{', close_ch:='}', block:=blk, eno=>ok);
	EXPECT_TRUE(ok);
	EXPECT_EQ_STRING('{"b":{"c":1}}', blk);
	// string value containing bracket is not counted as bracket
	src := '{"a":"x}y","b":{"v":1}}';
	json_find_block(dat:=src, key:='b', open_ch:='{', close_ch:='}', block:=blk, eno=>ok);
	EXPECT_TRUE(ok);
	EXPECT_EQ_STRING('{"v":1}', blk);
	// key not found: eno = false
	ok := true;
	json_find_block(dat:=src, key:='missing', open_ch:='{', close_ch:='}', block:=blk, eno=>ok);
	EXPECT_FALSE(ok);
	{end}
END_TEST_S

TEST_S(test_json_unescape)
	var
		res: string[JSON_STRING_SIZE];
		ok: bool;
	end
	{st}
	// --- Single escape sequences ---
	res := json_unescape('\"');
	EXPECT_EQ_STRING('"', res);
	res := json_unescape('\\');
	EXPECT_EQ_STRING('\', res);
	res := json_unescape('\t');
	EXPECT_EQ_STRING('$t', res);
	res := json_unescape('\n');
	EXPECT_EQ_STRING('$n', res);
	res := json_unescape('\r');
	EXPECT_EQ_STRING('$r', res);
	res := json_unescape('\b');
	EXPECT_EQ_STRING('$08', res);
	res := json_unescape('\f');
	EXPECT_EQ_STRING('$0c', res);
	res := json_unescape('\x41');
	EXPECT_EQ_STRING('A', res);
	res := json_unescape('\u0041');
	EXPECT_EQ_STRING('A', res);
	res := json_unescape('\u000a');
	EXPECT_EQ_STRING('$n', res);
	res := json_unescape('\u0022');
	EXPECT_EQ_STRING('"', res);
	// --- Multiple characters ---
	res := json_unescape('Hello');
	EXPECT_EQ_STRING('Hello', res);
	res := json_unescape('a\"b');
	EXPECT_EQ_STRING('a"b', res);
	res := json_unescape('a\tb');
	EXPECT_EQ_STRING('a$tb', res);
	res := json_unescape('a\nb');
	EXPECT_EQ_STRING('a$nb', res);
	res := json_unescape('a\"b\nc');
	EXPECT_EQ_STRING('a"b$nc', res);
	// --- Error cases: verify eno=false is returned ---
	res := json_unescape(in:='\', eno=>ok);      // trailing backslash
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\q', eno=>ok);     // unknown escape character
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\x4', eno=>ok);    // \x: insufficient hex digits
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\xGH', eno=>ok);   // \x: invalid hex digit
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\u00A', eno=>ok);  // \u: insufficient hex digits
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\u0080', eno=>ok); // \u: non-ASCII code (128)
	EXPECT_FALSE(ok);
	{end}
END_TEST_S
