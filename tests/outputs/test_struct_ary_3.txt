TEST_S(test_StructAry3_jsonstr_to_data)
	var
		json: string[1024];
		data: StructAry3_t;
	end
	{st}
	json := '{"units":[{"id":1,"body":{"pos":{"x":1.0,"y":2.0},"label":"unit1"}},{"id":2,"body":{"pos":{"x":3.0,"y":4.0},"label":"unit2"}}]}';
	StructAry3_jsonstr_to_data(json, data);
	EXPECT_EQ_DINT(1, data.units[0].id);
	EXPECT_NEAR(1.0, data.units[0].body.pos.x, 0.001);
	EXPECT_NEAR(2.0, data.units[0].body.pos.y, 0.001);
	EXPECT_EQ_STRING('unit1', data.units[0].body.label);
	EXPECT_EQ_DINT(2, data.units[1].id);
	EXPECT_NEAR(3.0, data.units[1].body.pos.x, 0.001);
	EXPECT_NEAR(4.0, data.units[1].body.pos.y, 0.001);
	EXPECT_EQ_STRING('unit2', data.units[1].body.label);
	{end}
END_TEST_S

TEST_S(test_json_bytes_find_key)
	var
		raw: array[0..127] of byte;
		wp: dint;
		idx: dint;
		ok: bool;
	end
	{st}
	// basic: single integer field
	// {"x":42} - key 'x' value starts at byte 5 ('4' = 52)
	wp := json_bytes_insert_str(json:=raw, pos:=0, str:='{"x":42}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(raw, 'x');
	EXPECT_EQ_DINT(5, idx);
	EXPECT_EQ_DINT(52, byte_to_dint(raw[idx]));
	// second field: {"a":1,"b":2} - key 'b' value starts at byte 11 ('2' = 50)
	wp := json_bytes_insert_str(json:=raw, pos:=0, str:='{"a":1,"b":2}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(raw, 'b');
	EXPECT_EQ_DINT(11, idx);
	EXPECT_EQ_DINT(50, byte_to_dint(raw[idx]));
	// nested: key inside nested object is not matched; top-level key is
	// {"a":{"key":1},"key":2} - top-level 'key' value at byte 21 ('2' = 50)
	wp := json_bytes_insert_str(json:=raw, pos:=0, str:='{"a":{"key":1},"key":2}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(raw, 'key');
	EXPECT_EQ_DINT(21, idx);
	EXPECT_EQ_DINT(50, byte_to_dint(raw[idx]));
	// not found: eno = false
	wp := json_bytes_insert_str(json:=raw, pos:=0, str:='{"other":1}', eno=>ok);
	raw[wp] := 0;
	idx := json_bytes_find_key(raw, 'x');
	EXPECT_EQ_DINT(-1, idx);
	{end}
END_TEST_S

	var
		src: string[JSON_STRING_SIZE];
		pos: dint;
	end
	{st}
	// basic: key found at depth 1
	src := '{"key":1}';
	pos := json_find_key(src, 'key');
	EXPECT_EQ_DINT(2, pos);
	// nested: key inside nested object is not matched; outer key is
	src := '{"outer":{"key":1},"key":2}';
	pos := json_find_key(src, 'key');
	EXPECT_EQ_DINT(20, pos);
	// string value containing escaped key pattern is skipped
	src := '{"ab":"\"key\":x","key":1}';
	pos := json_find_key(src, 'key');
	EXPECT_EQ_DINT(19, pos);
	// not found returns 0
	src := '{"other":1}';
	pos := json_find_key(src, 'key');
	EXPECT_EQ_DINT(0, pos);
	{end}

TEST_S(test_json_unescape)
	var
		res: string[JSON_STRING_SIZE];
		ok: bool;
	end
	{st}
	// --- Single escape sequences ---
	res := json_unescape('\"');
	EXPECT_EQ_STRING('"', res);
	res := json_unescape('\\');
	EXPECT_EQ_STRING('\', res);
	res := json_unescape('\t');
	EXPECT_EQ_STRING('$t', res);
	res := json_unescape('\n');
	EXPECT_EQ_STRING('$n', res);
	res := json_unescape('\r');
	EXPECT_EQ_STRING('$r', res);
	res := json_unescape('\b');
	EXPECT_EQ_STRING('$08', res);
	res := json_unescape('\f');
	EXPECT_EQ_STRING('$0c', res);
	res := json_unescape('\x41');
	EXPECT_EQ_STRING('A', res);
	res := json_unescape('\u0041');
	EXPECT_EQ_STRING('A', res);
	res := json_unescape('\u000a');
	EXPECT_EQ_STRING('$n', res);
	res := json_unescape('\u0022');
	EXPECT_EQ_STRING('"', res);
	// --- Multiple characters ---
	res := json_unescape('Hello');
	EXPECT_EQ_STRING('Hello', res);
	res := json_unescape('a\"b');
	EXPECT_EQ_STRING('a"b', res);
	res := json_unescape('a\tb');
	EXPECT_EQ_STRING('a$tb', res);
	res := json_unescape('a\nb');
	EXPECT_EQ_STRING('a$nb', res);
	res := json_unescape('a\"b\nc');
	EXPECT_EQ_STRING('a"b$nc', res);
	// --- Error cases: verify eno=false is returned ---
	res := json_unescape(in:='\', eno=>ok);      // trailing backslash
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\q', eno=>ok);     // unknown escape character
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\x4', eno=>ok);    // \x: insufficient hex digits
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\xGH', eno=>ok);   // \x: invalid hex digit
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\u00A', eno=>ok);  // \u: insufficient hex digits
	EXPECT_FALSE(ok);
	res := json_unescape(in:='\u0080', eno=>ok); // \u: non-ASCII code (128)
	EXPECT_FALSE(ok);
	{end}
END_TEST_S
