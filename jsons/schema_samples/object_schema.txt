function json_escape: string[JSON_STRING_SIZE]
	_DEFINE_ENENO_()
	var_input in: string[JSON_STRING_SIZE]; end_var
	var_temp
		result: string[JSON_STRING_SIZE];
		ci, src_len: dint;
		ch: string[8];
	end_var
(*{st}*)
	result := '';
	src_len := len(in);
	for ci := 1 to src_len by 1 do
		ch := mid(in, 1, dint_to_uint(ci));
		if      ch = '"' then result := concat(result, '\"');
		elsif   ch = '\' then result := concat(result, '\\');
		elsif   ch = '$t'  then result := concat(result, '\t');
		elsif   ch = '$n'  then result := concat(result, '\n');
		elsif   ch = '$r'  then result := concat(result, '\r');
		elsif   ch = '$08' then result := concat(result, '\b');
		elsif   ch = '$0c' then result := concat(result, '\f');
		elsif   ch = '$00' then result := concat(result, '\u0000');
		elsif   ch = '$01' then result := concat(result, '\u0001');
		elsif   ch = '$02' then result := concat(result, '\u0002');
		elsif   ch = '$03' then result := concat(result, '\u0003');
		elsif   ch = '$04' then result := concat(result, '\u0004');
		elsif   ch = '$05' then result := concat(result, '\u0005');
		elsif   ch = '$06' then result := concat(result, '\u0006');
		elsif   ch = '$07' then result := concat(result, '\u0007');
		elsif   ch = '$0b' then result := concat(result, '\u000b');
		elsif   ch = '$0e' then result := concat(result, '\u000e');
		elsif   ch = '$0f' then result := concat(result, '\u000f');
		elsif   ch = '$10' then result := concat(result, '\u0010');
		elsif   ch = '$11' then result := concat(result, '\u0011');
		elsif   ch = '$12' then result := concat(result, '\u0012');
		elsif   ch = '$13' then result := concat(result, '\u0013');
		elsif   ch = '$14' then result := concat(result, '\u0014');
		elsif   ch = '$15' then result := concat(result, '\u0015');
		elsif   ch = '$16' then result := concat(result, '\u0016');
		elsif   ch = '$17' then result := concat(result, '\u0017');
		elsif   ch = '$18' then result := concat(result, '\u0018');
		elsif   ch = '$19' then result := concat(result, '\u0019');
		elsif   ch = '$1a' then result := concat(result, '\u001a');
		elsif   ch = '$1b' then result := concat(result, '\u001b');
		elsif   ch = '$1c' then result := concat(result, '\u001c');
		elsif   ch = '$1d' then result := concat(result, '\u001d');
		elsif   ch = '$1e' then result := concat(result, '\u001e');
		elsif   ch = '$1f' then result := concat(result, '\u001f');
		else result := concat(result, ch);
		end_if;
	end_for;
	json_escape := result;
(*{end}*)
end_function

function json_char_from_asciicharcode: string[8]
	_DEFINE_ENENO_()
	var_input code: dint; end_var
	var_temp
		printable: string[128] := ' !"#$$%&$27()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~';
	end_var
(*{st}*)
	case code of
		32..126: json_char_from_asciicharcode := mid(printable, 1, dint_to_uint(code - 31));
		0:   json_char_from_asciicharcode := '$00';
		1:   json_char_from_asciicharcode := '$01';
		2:   json_char_from_asciicharcode := '$02';
		3:   json_char_from_asciicharcode := '$03';
		4:   json_char_from_asciicharcode := '$04';
		5:   json_char_from_asciicharcode := '$05';
		6:   json_char_from_asciicharcode := '$06';
		7:   json_char_from_asciicharcode := '$07';
		8:   json_char_from_asciicharcode := '$08';
		9:   json_char_from_asciicharcode := '$t';
		10:  json_char_from_asciicharcode := '$n';
		11:  json_char_from_asciicharcode := '$0b';
		12:  json_char_from_asciicharcode := '$0c';
		13:  json_char_from_asciicharcode := '$r';
		14:  json_char_from_asciicharcode := '$0e';
		15:  json_char_from_asciicharcode := '$0f';
		16:  json_char_from_asciicharcode := '$10';
		17:  json_char_from_asciicharcode := '$11';
		18:  json_char_from_asciicharcode := '$12';
		19:  json_char_from_asciicharcode := '$13';
		20:  json_char_from_asciicharcode := '$14';
		21:  json_char_from_asciicharcode := '$15';
		22:  json_char_from_asciicharcode := '$16';
		23:  json_char_from_asciicharcode := '$17';
		24:  json_char_from_asciicharcode := '$18';
		25:  json_char_from_asciicharcode := '$19';
		26:  json_char_from_asciicharcode := '$1a';
		27:  json_char_from_asciicharcode := '$1b';
		28:  json_char_from_asciicharcode := '$1c';
		29:  json_char_from_asciicharcode := '$1d';
		30:  json_char_from_asciicharcode := '$1e';
		31:  json_char_from_asciicharcode := '$1f';
		127: json_char_from_asciicharcode := '$7f';
	else
		eno := false; // Invalid code
		json_char_from_asciicharcode := '';
	end_case;
(*{end}*)
end_function

function json_bytes_insert_str: dint
	_DEFINE_ENENO_()
	var_in_out json: array[*] of byte; end_var
	var_input pos: dint; end_var
	var_input str: string[JSON_STRING_SIZE]; end_var
	var_temp
		slen, wp, ci, code: dint;
	end_var
(*{st}*)
	slen := len(str);
	wp := pos;
	ci := 0;
	while ci < slen do
		code := json_asciicharcode(in:=mid(str, 1, dint_to_uint(ci + 1)), eno=>eno);
		if not eno then
			json_bytes_insert_str := -1;
			return;
		end_if;
		json[wp] := dint_to_byte(code);
		wp := wp + 1;
		ci := ci + 1;
	end_while;
	json_bytes_insert_str := wp;
(*{end}*)
end_function

function json_asciicharcode: dint
	_DEFINE_ENENO_()
	var_input in: string[2]; end_var
	var_temp
		printable: string[128] := ' !"#$$%&$27()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~';
	end_var
(*{st}*)
	json_asciicharcode := find(printable, in) + 31;
	if json_asciicharcode <= 31 then
		if      in = '$08' then json_asciicharcode := 8;
		elsif   in = '$t'  then json_asciicharcode := 9;
		elsif   in = '$n'  then json_asciicharcode := 10;
		elsif   in = '$0b' then json_asciicharcode := 11;
		elsif   in = '$0c' then json_asciicharcode := 12;
		elsif   in = '$r'  then json_asciicharcode := 13;
		else json_asciicharcode := 0;
		end_if;
	end_if;
(*{end}*)
end_function

function json_find_key : dint
	var_in_out dat: string[JSON_STRING_SIZE]; end_var
	var_input key: string[JSON_STRING_SIZE]; end_var
	var_temp
		ci, slen, klen, depth: dint;
		ch: string[2];
	end_var
(*{st}*)
	// Scan dat at nesting depth 1, skipping string literals and nested blocks.
	// Returns the 1-based position of the leading '"' of '"key":', or 0 if not found.
	slen := len(dat);
	klen := len(key) + 3; // len('"' + key + '":')
	depth := 0;
	ci := 1;
	while ci <= slen do
		ch := mid(dat, 1, dint_to_uint(ci));
		if ch = '"' then
			if depth = 1 and mid(dat, dint_to_uint(klen), dint_to_uint(ci)) = concat('"', concat(key, '":')) then
				json_find_key := ci;
				return;
			end_if;
			// skip string literal
			ci := ci + 1;
			while ci <= slen do
				ch := mid(dat, 1, dint_to_uint(ci));
				if ch = '\' then ci := ci + 1;
				elsif ch = '"' then exit;
				end_if;
				ci := ci + 1;
			end_while;
		elsif ch = '{' or ch = '[' then
			depth := depth + 1;
		elsif ch = '}' or ch = ']' then
			depth := depth - 1;
		end_if;
		ci := ci + 1;
	end_while;
	json_find_key := 0;
(*{end}*)
end_function

function json_bytes_find_key: dint
	_DEFINE_ENENO_()
	var_in_out dat: array[*] of byte; end_var
	var_input key: string[JSON_STRING_SIZE]; end_var
	var_temp
		ci, klen, klen_i, tmp_ci: dint;
		depth: dint;
		sv, kpat: string[JSON_STRING_SIZE];
		match: bool;
	end_var
(*{st}*)
	// Scan null-terminated byte array at nesting depth 1, skipping string literals.
	// Returns the 0-based byte index of the value (after '"key":'), or returns -1 if not found.
	kpat := concat('"', concat(key, '":'));
	klen := len(kpat);
	depth := 0;
	ci := 0;
	while dat[ci] <> 0 do
		if dat[ci] = 34 then  // '"'
			if depth = 1 then
				sv := '';
				tmp_ci := ci;
				match := true;
				klen_i := 0;
				while klen_i < klen and match do
					sv := concat(sv, json_char_from_asciicharcode(code:=byte_to_dint(dat[tmp_ci]), eno=>eno));
					if not eno then match := false; eno := true; end_if;
					tmp_ci := tmp_ci + 1;
					klen_i := klen_i + 1;
				end_while;
				if match and sv = kpat then
					json_bytes_find_key := ci + klen;
					return;
				end_if;
			end_if;
			// skip string literal
			ci := ci + 1;
			while dat[ci] <> 34 do
				if dat[ci] = 92 then ci := ci + 1; end_if;  // 92 = '\'
				ci := ci + 1;
			end_while;
		elsif dat[ci] = 123 or dat[ci] = 91 then  // '{' or '['
			depth := depth + 1;
		elsif dat[ci] = 125 or dat[ci] = 93 then  // '}' or ']'
			depth := depth - 1;
		end_if;
		ci := ci + 1;
	end_while;
	json_bytes_find_key := -1;
(*{end}*)
end_function

function json_unescape: string[JSON_STRING_SIZE]
	_DEFINE_ENENO_()
	var_input in: string[JSON_STRING_SIZE]; end_var
	var_temp
		result: string[JSON_STRING_SIZE];
		ci, src_len: dint;
		ch: string[2];
		h1, h2, h3, h4: string[2];
		hex_d: string[23] := '0123456789abcdefABCDEF';
		i1, i2, i3, i4, code: dint;
	end_var
(*{st}*)
	result := '';
	ci := 1;
	src_len := len(in);
	while ci <= src_len do
		ch := mid(in, 1, dint_to_uint(ci));
		if ch = '\' then
			ci := ci + 1;
			if ci > src_len then eno := false; return; end_if;
			ch := mid(in, 1, dint_to_uint(ci));
			if      ch = '"' then result := concat(result, '"');
			elsif   ch = '\' then result := concat(result, '\');
			elsif   ch = 't'  then result := concat(result, '$t');
			elsif   ch = 'n'  then result := concat(result, '$n');
			elsif   ch = 'r'  then result := concat(result, '$r');
			elsif   ch = 'b'  then result := concat(result, '$08');
			elsif   ch = 'f'  then result := concat(result, '$0c');
			elsif   ch = 'x'  then
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h1 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h2 := mid(in, 1, dint_to_uint(ci));
				i1 := find(hex_d, h1) - 1;
				if i1 >= 16 then i1 := i1 - 6; end_if;
				if i1 < 0 then eno := false; return; end_if;
				i2 := find(hex_d, h2) - 1;
				if i2 >= 16 then i2 := i2 - 6; end_if;
				if i2 < 0 then eno := false; return; end_if;
				code := i1 * 16 + i2;
				result := concat(result, json_char_from_asciicharcode(code:=code, eno=>eno));
				if not eno then return; end_if;
			elsif   ch = 'u'  then
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h1 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h2 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h3 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h4 := mid(in, 1, dint_to_uint(ci));
				i1 := find(hex_d, h1) - 1;
				if i1 >= 16 then i1 := i1 - 6; end_if;
				if i1 < 0 then eno := false; return; end_if;
				i2 := find(hex_d, h2) - 1;
				if i2 >= 16 then i2 := i2 - 6; end_if;
				if i2 < 0 then eno := false; return; end_if;
				i3 := find(hex_d, h3) - 1;
				if i3 >= 16 then i3 := i3 - 6; end_if;
				if i3 < 0 then eno := false; return; end_if;
				i4 := find(hex_d, h4) - 1;
				if i4 >= 16 then i4 := i4 - 6; end_if;
				if i4 < 0 then eno := false; return; end_if;
				code := i1 * 4096 + i2 * 256 + i3 * 16 + i4;
				result := concat(result, json_char_from_asciicharcode(code:=code, eno=>eno));
				if not eno then return; end_if;
			else
				eno := false;
				return;
			end_if;
		else
			result := concat(result, ch);
		end_if;
		ci := ci + 1;
	end_while;
	json_unescape := result;
(*{end}*)
end_function

function JsonRoot_jsonstr_to_data
	_DEFINE_ENENO_()
	var_in_out json: string[1024]; end_var
	var_in_out data: JsonRoot_t; end_var
	var_temp
		pos, qos, qp: dint;
		pos1, pos2: dint;
		jblk1: string[1024];
		jblk1_len: dint;
		jblk1_sub: string[1024];
		jblk2: string[1024];
		jblk2_len: dint;
		jblk2_sub: string[1024];
		depth: dint;
	end_var
(*{st}*)
	// name: string
	pos := json_find_key(json, 'name');
	if pos = 0 then eno := false; return; end_if;
	pos := pos + 8;
	qos := pos;
	while qos <= uint_to_dint(len(json)) do
		if mid(json, 1, dint_to_uint(qos)) = '\' then qos := qos + 2;
		elsif mid(json, 1, dint_to_uint(qos)) = '"' then exit;
		else qos := qos + 1;
		end_if;
	end_while;
	data.name := json_unescape(in:=mid(json, dint_to_uint(qos - pos), dint_to_uint(pos)), eno=>eno);
	if not eno then return; end_if;
	// pos: struct
	pos := json_find_key(json, 'pos');
	if pos = 0 then eno := false; return; end_if;
	pos := pos + 6;
	qos := pos;
	depth := 1;
	while depth > 0 do
		qos := qos + 1;
		if mid(json, 1, dint_to_uint(qos)) = '"' then
			// skip string literal to avoid false bracket matches
			qos := qos + 1;
			while mid(json, 1, dint_to_uint(qos)) <> '"' do
				if mid(json, 1, dint_to_uint(qos)) = '\' then qos := qos + 1; end_if;
				qos := qos + 1;
			end_while;
		elsif mid(json, 1, dint_to_uint(qos)) = '{' then depth := depth + 1;
