function json_bytes_insert_text: dint
	_DEFINE_ENENO_()
	var_in_out dat: array[*] of byte; end_var

	var_input str: string[JSON_STRING_SIZE]; end_var
	var_temp
		slen, wp, ci, code: dint;
	end_var
	slen := len(str);
	wp := index;
	ci := 0;
	while ci < slen do
		code := json_asciicharcode(in:=mid(str, 1, dint_to_uint(ci + 1)), eno=>eno);
		if not eno then
			json_bytes_insert_text := -1;
			return;
		end_if;
		dat[wp] := dint_to_byte(code);
		wp := wp + 1;
		ci := ci + 1;
	end_while;
	json_bytes_insert_text := wp;
end_function

function json_asciicharcode: dint
	_DEFINE_ENENO_()
	var_input in: string[2]; end_var
	var_temp
		printable: string[128] := ' !"#$$%&$27()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~';
	end_var
	json_asciicharcode := find(printable, in) + 31;
	if json_asciicharcode <= 31 then
		if      in = '$08' then json_asciicharcode := 8;
		elsif   in = '$t'  then json_asciicharcode := 9;
		elsif   in = '$n'  then json_asciicharcode := 10;
		elsif   in = '$0b' then json_asciicharcode := 11;
		elsif   in = '$0c' then json_asciicharcode := 12;
		elsif   in = '$r'  then json_asciicharcode := 13;
		else json_asciicharcode := 0;
		end_if;
	end_if;
end_function

function json_find_key : dint
	var_in_out dat: string[JSON_STRING_SIZE]; end_var
	var_input key: string[JSON_STRING_SIZE]; end_var
	var_temp
		ci, slen, klen, depth: dint;
		ch: string[2];
	end_var
	// Scan dat at nesting depth 1, skipping string literals and nested blocks.
	// Returns the 1-based position of the leading '"' of '"key":', or 0 if not found.
	slen := len(dat);
	klen := len(key) + 3; // len('"' + key + '":')
	depth := 0;
	ci := 1;
	while ci <= slen do
		ch := mid(dat, 1, dint_to_uint(ci));
		if ch = '"' then
			if depth = 1 and mid(dat, dint_to_uint(klen), dint_to_uint(ci)) = concat('"', concat(key, '":')) then
				json_find_key := ci;
				return;
			end_if;
			// skip string literal
			ci := ci + 1;
			while ci <= slen do
				ch := mid(dat, 1, dint_to_uint(ci));
				if ch = '\' then ci := ci + 1;
				elsif ch = '"' then exit;
				end_if;
				ci := ci + 1;
			end_while;
		elsif ch = '{' or ch = '[' then
			depth := depth + 1;
		elsif ch = '}' or ch = ']' then
			depth := depth - 1;
		end_if;
		ci := ci + 1;
	end_while;
	json_find_key := 0;
end_function

function json_bytes_find_key: dint
	_DEFINE_ENENO_()
	var_in_out dat: array[*] of byte; end_var
	var_in_out key: array[*] of byte; end_var
	var_input start_index: dint; end_var
	var_input end_index: dint := -1; end_var
	var_temp
		ci, klen_i, tmp_ci: dint;
		depth, klen: dint;
		match: bool;
		ei: dint;
	end_var
	// Scan dat[start_index..end_index] at nesting depth 1, skipping string literals.
	// Returns the 0-based byte index of the value (after '"key":'), or -1 if not found.
	// end_index=-1 means scan to the end of dat.
	if end_index < 0 then ei := upper_bound(dat, 1); else ei := end_index; end_if;
	klen := upper_bound(key, 1) - lower_bound(key, 1) + 1;
	depth := 0;
	ci := start_index;
	while ci <= ei and dat[ci] <> 0 do
		if dat[ci] = 34 then  // '"'
			if depth = 1 then
				match := true;
				tmp_ci := ci + 1;
				klen_i := 0;
				while klen_i < klen and match and tmp_ci <= ei do
					if dat[tmp_ci] <> key[klen_i] then match := false; end_if;
					tmp_ci := tmp_ci + 1;
					klen_i := klen_i + 1;
				end_while;
				if match and tmp_ci + 1 <= ei and dat[tmp_ci] = 34 and dat[tmp_ci + 1] = 58 then  // '"' and ':'
					json_bytes_find_key := tmp_ci + 2;
					return;
				end_if;
			end_if;
			// skip string literal
			ci := ci + 1;
			while ci <= ei and dat[ci] <> 34 do
				if dat[ci] = 92 then
					ci := ci + 1;
					if ci > ei then json_bytes_find_key := -1; return; end_if;
				end_if;
				ci := ci + 1;
			end_while;
			if ci > ei then json_bytes_find_key := -1; return; end_if;
		elsif dat[ci] = 123 or dat[ci] = 91 then  // '{' or '['
			depth := depth + 1;
		elsif dat[ci] = 125 or dat[ci] = 93 then  // '}' or ']'
			depth := depth - 1;
		end_if;
		ci := ci + 1;
	end_while;
	json_bytes_find_key := -1;
end_function

function json_find_block
	_DEFINE_ENENO_()
	var_in_out dat: string[JSON_STRING_SIZE]; end_var
	var_input key: string[JSON_STRING_SIZE]; end_var
	var_input open_ch: string[2]; end_var
	var_input close_ch: string[2]; end_var
	var_in_out block: string[JSON_STRING_SIZE]; end_var
	var_temp
		pos, qos, slen, depth: dint;
		ch: string[2];
	end_var
	// Find key at depth 1 in dat, then extract the following {..} or [..] block.
	// Writes the block into block. Sets eno=false if key not found or brackets unmatched.
	pos := json_find_key(dat:=dat, key:=key);
	if pos = 0 then eno := false; return; end_if;
	pos := pos + uint_to_dint(len(key)) + 3;  // skip '"key":'
	qos := pos;
	slen := uint_to_dint(len(dat));
	depth := 1;
	while depth > 0 do
		qos := qos + 1;
		if qos > slen then eno := false; return; end_if;
		ch := mid(dat, 1, dint_to_uint(qos));
		if ch = '"' then
			// skip string literal to avoid false bracket matches
			qos := qos + 1;
			while mid(dat, 1, dint_to_uint(qos)) <> '"' do
				if mid(dat, 1, dint_to_uint(qos)) = '\' then qos := qos + 1; end_if;
				qos := qos + 1;
			end_while;
		elsif ch = open_ch then depth := depth + 1;
		elsif ch = close_ch then depth := depth - 1;
		end_if;
	end_while;
	block := mid(dat, dint_to_uint(qos - pos + 1), dint_to_uint(pos));
end_function

function json_extract_scalar
	_DEFINE_ENENO_()
	var_in_out dat: string[JSON_STRING_SIZE]; end_var
	var_input key: string[JSON_STRING_SIZE]; end_var
	var_in_out sv: string[JSON_STRING_SIZE]; end_var
	var_temp
		pos, qp, slen: dint;
		tail: string[JSON_STRING_SIZE];
	end_var
	// Find key at depth 1 in dat, then extract the scalar value (up to ',', '}', or ']').
	// Writes the value into sv. Sets eno=false if key not found or no delimiter.
	pos := json_find_key(dat:=dat, key:=key);
	if pos = 0 then eno := false; return; end_if;
	pos := pos + uint_to_dint(len(key)) + 3;  // skip '"key":'
	slen := uint_to_dint(len(dat));
	tail := mid(dat, dint_to_uint(slen - pos + 1), dint_to_uint(pos));
	qp := find(tail, ',');
	if qp = 0 then qp := find(tail, '}'); end_if;
	if qp = 0 then qp := find(tail, ']'); end_if;
	if qp = 0 then eno := false; return; end_if;
	sv := mid(dat, dint_to_uint(qp - 1), dint_to_uint(pos));
end_function

function json_bytes_find_block
	_DEFINE_ENENO_()
	var_in_out dat: array[*] of byte; end_var
	var_in_out key: array[*] of byte; end_var
	var_input open_byte: byte; end_var
	var_input close_byte: byte; end_var
	var_input start_index: dint; end_var
	var_input end_index: dint; end_var
	var_in_out block_start_index: dint; end_var
	var_in_out block_end_index: dint; end_var
	var_temp
		raw_ci, raw_qos, raw_depth: dint;
		dat_ub: dint;
	end_var
	// Find key in byte array dat and set block_start_index/block_end_index
	// to the inclusive range [open_byte..close_byte] within dat.
	// open_byte/close_byte are the bracket bytes (e.g. 123/125 for {}).
	// Sets eno=false if key not found or brackets unmatched.
	raw_ci := json_bytes_find_key(dat:=dat, key:=key, start_index:=start_index, end_index:=end_index, eno=>eno);
	if raw_ci = -1 then eno := false; return; end_if;
	raw_qos := raw_ci;
	raw_depth := 1;
	dat_ub := upper_bound(dat, 1);
	while raw_depth > 0 do
		raw_qos := raw_qos + 1;
		if raw_qos > dat_ub then eno := false; return; end_if;
		if dat[raw_qos] = 34 then  // 34 = '"'
			// skip string literal
			raw_qos := raw_qos + 1;
			while raw_qos <= dat_ub and dat[raw_qos] <> 34 do
				if dat[raw_qos] = 92 then
					raw_qos := raw_qos + 1;
					if raw_qos > dat_ub then eno := false; return; end_if;
				end_if;
				raw_qos := raw_qos + 1;
			end_while;
			if raw_qos > dat_ub then eno := false; return; end_if;
		elsif dat[raw_qos] = open_byte then raw_depth := raw_depth + 1;
		elsif dat[raw_qos] = close_byte then raw_depth := raw_depth - 1;
		end_if;
	end_while;
	block_start_index := raw_ci;
	block_end_index := raw_qos;
end_function

function json_bytes_extract_scalar
	_DEFINE_ENENO_()
	var_in_out dat: array[*] of byte; end_var
	var_in_out key: array[*] of byte; end_var
	var_input start_index: dint; end_var
	var_input end_index: dint; end_var
	var_in_out sv: string[JSON_STRING_SIZE]; end_var
	var_temp
		raw_ci: dint;
		dat_ub: dint;
	end_var
	// Find key in byte array dat, then extract the scalar value (up to ',', '}', ']', or NUL).
	// Writes the value into sv. Sets eno=false if key not found.
	raw_ci := json_bytes_find_key(dat:=dat, key:=key, start_index:=start_index, end_index:=end_index, eno=>eno);
	if raw_ci = -1 then eno := false; return; end_if;
	dat_ub := upper_bound(dat, 1);
	sv := '';
	while raw_ci <= dat_ub and dat[raw_ci] <> 44 and dat[raw_ci] <> 125 and dat[raw_ci] <> 93 and dat[raw_ci] <> 0 do
		sv := concat(sv, json_char_from_asciicharcode(code:=byte_to_dint(dat[raw_ci]), eno=>eno));
		if not eno then return; end_if;
		raw_ci := raw_ci + 1;
	end_while;
	if raw_ci > dat_ub then eno := false; return; end_if;
end_function

function json_bytes_to_str
	_DEFINE_ENENO_()
	var_in_out dat: array[*] of byte; end_var
	var_input start_index: dint; end_var
	var_input end_index: dint; end_var
	var_in_out sv: string[JSON_STRING_SIZE]; end_var
	var_temp
		raw_ci: dint;
	end_var
	// Convert dat[start_index..end_index] into a string.
	// Sets eno=false if a byte cannot be converted.
	if end_index > upper_bound(dat, 1) then eno := false; return; end_if;
	sv := '';
	raw_ci := start_index;
	while raw_ci <= end_index do
		sv := concat(sv, json_char_from_asciicharcode(code:=byte_to_dint(dat[raw_ci]), eno=>eno));
		if not eno then return; end_if;
		raw_ci := raw_ci + 1;
	end_while;
end_function

function json_unescape: string[JSON_STRING_SIZE]
	_DEFINE_ENENO_()
	var_input in: string[JSON_STRING_SIZE]; end_var
	var_temp
		result: string[JSON_STRING_SIZE];
		ci, src_len: dint;
		ch: string[2];
		h1, h2, h3, h4: string[2];
		hex_d: string[23] := '0123456789abcdefABCDEF';
		i1, i2, i3, i4, code: dint;
	end_var
	result := '';
	ci := 1;
	src_len := len(in);
	while ci <= src_len do
		ch := mid(in, 1, dint_to_uint(ci));
		if ch = '\' then
			ci := ci + 1;
			if ci > src_len then eno := false; return; end_if;
			ch := mid(in, 1, dint_to_uint(ci));
			if      ch = '"' then result := concat(result, '"');
			elsif   ch = '\' then result := concat(result, '\');
			elsif   ch = 't'  then result := concat(result, '$t');
			elsif   ch = 'n'  then result := concat(result, '$n');
			elsif   ch = 'r'  then result := concat(result, '$r');
			elsif   ch = 'b'  then result := concat(result, '$08');
			elsif   ch = 'f'  then result := concat(result, '$0c');
			elsif   ch = 'x'  then
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h1 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h2 := mid(in, 1, dint_to_uint(ci));
				i1 := find(hex_d, h1) - 1;
				if i1 >= 16 then i1 := i1 - 6; end_if;
				if i1 < 0 then eno := false; return; end_if;
				i2 := find(hex_d, h2) - 1;
				if i2 >= 16 then i2 := i2 - 6; end_if;
				if i2 < 0 then eno := false; return; end_if;
				code := i1 * 16 + i2;
				result := concat(result, json_char_from_asciicharcode(code:=code, eno=>eno));
				if not eno then return; end_if;
			elsif   ch = 'u'  then
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h1 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h2 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h3 := mid(in, 1, dint_to_uint(ci));
				ci := ci + 1;
				if ci > src_len then eno := false; return; end_if;
				h4 := mid(in, 1, dint_to_uint(ci));
				i1 := find(hex_d, h1) - 1;
				if i1 >= 16 then i1 := i1 - 6; end_if;
				if i1 < 0 then eno := false; return; end_if;
				i2 := find(hex_d, h2) - 1;
				if i2 >= 16 then i2 := i2 - 6; end_if;
				if i2 < 0 then eno := false; return; end_if;
				i3 := find(hex_d, h3) - 1;
				if i3 >= 16 then i3 := i3 - 6; end_if;
				if i3 < 0 then eno := false; return; end_if;
				i4 := find(hex_d, h4) - 1;
				if i4 >= 16 then i4 := i4 - 6; end_if;
				if i4 < 0 then eno := false; return; end_if;
				code := i1 * 4096 + i2 * 256 + i3 * 16 + i4;
				result := concat(result, json_char_from_asciicharcode(code:=code, eno=>eno));
				if not eno then return; end_if;
			else
				eno := false;
				return;
			end_if;
		else
			result := concat(result, ch);
		end_if;
		ci := ci + 1;
	end_while;
	json_unescape := result;
end_function
	
function JsonRoot_iecdata_to_jsontext
	_DEFINE_ENENO_()
	var_in_out data: JsonRoot_t; end_var
	var_in_out txt: string[JSON_STRING_SIZE]; end_var
	var_temp
		sv: string[256];
	end_var
	txt := '{';
	txt := concat(txt, '"matrix":');
	txt := concat(txt, '[');
	txt := concat(txt, '[');
	sv := dint_to_string(data.matrix[0, 0]);
	txt := concat(txt, sv);
	txt := concat(txt, ',');
	sv := dint_to_string(data.matrix[0, 1]);
	txt := concat(txt, sv);
	txt := concat(txt, ',');
	sv := dint_to_string(data.matrix[0, 2]);
